import bpy
import math

def setup_moon_scene():
    # 1. Get the Moon object 
    moon = bpy.data.objects.get("Moon")
    
    if not moon:
        print("Error: Object named 'Moon' not found!")
        return

    # 2. Place the moon perfectly in the center
    moon.location = (0.0, 0.0, 0.0)
    
    # 3. Setup the Moon's rotation for exactly 15 seconds
    moon.animation_data_clear() 
    moon.rotation_mode = 'XYZ'
    
    fps = bpy.context.scene.render.fps
    fps_base = bpy.context.scene.render.fps_base
    actual_fps = fps / fps_base
    total_frames = int(actual_fps * 15)
    
    bpy.context.scene.frame_start = 1
    bpy.context.scene.frame_end = total_frames
    
    moon.rotation_euler = (0, 0, 0)
    moon.keyframe_insert(data_path="rotation_euler", index=2, frame=1)
    
    moon.rotation_euler = (0, 0, math.radians(360))
    moon.keyframe_insert(data_path="rotation_euler", index=2, frame=total_frames + 1) 
    
    # Make the rotation constant
    if moon.animation_data and moon.animation_data.action:
        action = moon.animation_data.action
        if hasattr(action, 'fcurves'):
            for fcurve in action.fcurves:
                if fcurve.data_path == 'rotation_euler':
                    for kf in fcurve.keyframe_points:
                        kf.interpolation = 'LINEAR'
        elif hasattr(action, 'fcurve_ensure_for_datablock'):
            fcurve = action.fcurve_ensure_for_datablock(datablock=moon, data_path="rotation_euler", index=2)
            if fcurve:
                for kf in fcurve.keyframe_points:
                    kf.interpolation = 'LINEAR'

    # 4. Set up the Camera (ZOOMED OUT A BIT)
    camera = bpy.data.objects.get("Camera")
    if camera:
        bpy.context.scene.camera = camera
        camera.rotation_mode = 'XYZ'
        camera.rotation_euler = (math.radians(90), 0, 0)
        # Changed from -5.0 to -7.5 to give it some breathing room
        camera.location = (0.0, -7.5, 0.0) 
        
    # 5. Make the space background pitch black
    world = bpy.context.scene.world
    if world and world.use_nodes:
        bg_node = world.node_tree.nodes.get('Background')
        if bg_node:
            bg_node.inputs[0].default_value = (0, 0, 0, 1) 
            
    # 6. Set up the Light
    light = bpy.data.objects.get("Light")
    if light:
        # Move light back with the camera
        light.location = (0.0, -7.5, 0.0)
        light.rotation_mode = 'XYZ'
        light.rotation_euler = (math.radians(90), 0, 0)
        
        if hasattr(light.data, 'energy'):
            if light.data.type == 'POINT':
                light.data.energy = 5000 
            elif light.data.type == 'SUN':
                light.data.energy = 5    

setup_moon_scene()
